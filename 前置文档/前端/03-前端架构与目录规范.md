## 3. 分层职责

| 层 | 职责 | 禁止 |
|---|---|---|
| Page | 参数解析、登录拦截、导航、弹层状态、页面级 loading | 直接 `wx.request`、写业务校验 |
| Pane | 一个页签的完整数据流与列表逻辑 | 处理跨页导航（用事件抛给页面） |
| Component | 纯展示 + 局部交互，数据从 `properties` 进、事件出 | 读 store、发请求、`wx.navigateTo` |
| services | 接口封装、token 注入、重试、上传 | 处理 UI 反馈 |
| store | 跨页共享状态（user/ui/feed） | 存列表全量（页面自己维护） |
| utils | 纯函数与平台封装 | 依赖页面实例（`ui.js` 例外） |

## 4. 请求层 v2（1.0 核心改造）

### 4.1 行为规格

| 场景 | 行为 |
|---|---|
| 正常 | 注入 `Authorization`，返回 `data`，记录 `requestId` 与耗时 |
| `body.code !== 0` | 拒绝并抛 `{type:'BIZ', code, message, requestId}` |
| HTTP 401 | 清 token → 静默重登（`wx.login` + A01）→ **重放 1 次**；仍 401 才清会话跳登录 |
| HTTP 429 | Toast「操作太频繁，请稍后再试」，不重试 |
| 5xx / 网络失败 | GET 自动重试 1 次（间隔 800ms），仍失败抛 `NETWORK` |
| 耗时 >2s | `log.warn('request', url, ms)` |
| 失败 | 触发 `api_error` 埋点（节流：同 api 60s 内 ≤3 条） |

### 4.2 代码骨架

```js
// services/request.js（1.0）
const env = require('../config/env')
const mock = require('../mock/index')
const log = require('../utils/log')
const report = require('../utils/report')

let refreshing = null          // Promise | null，用于并发 401 去重
let lastUnauthorizedAt = 0

function err(type, code, message, requestId) {
  return { type, code: code || 0, message: message || '请求失败', requestId: requestId || '' }
}

function getToken() {
  return require('./auth').getToken()
}

/** 静默重登：wx.login → A01，失败返回 null */
function relogin() {
  if (refreshing) return refreshing
  refreshing = new Promise(resolve => {
    wx.login({
      success(res) {
        if (!res.code) return resolve(null)
        wx.request({
          url: env.baseURL + '/auth/wechat-login',
          method: 'POST',
          data: { code: res.code },
          timeout: env.TIMEOUT,
          success(r) {
            const b = r.data || {}
            if (b.code === 0 && b.data && b.data.token) {
              require('./auth').setSession(b.data.token, b.data.user)
              resolve(b.data.token)
            } else resolve(null)
          },
          fail: () => resolve(null)
        })
      },
      fail: () => resolve(null)
    })
  }).then(t => { refreshing = null; return t })
  return refreshing
}

function realRequest(opts, retried, relogged) {
  return new Promise((resolve, reject) => {
    const header = { 'content-type': 'application/json' }
    const token = getToken()
    if (opts.auth !== false && token) header.Authorization = 'Bearer ' + token

    const startedAt = Date.now()
    wx.request({
      url: env.baseURL + opts.url,
      method: opts.method || 'GET',
      data: opts.data || {},
      header,
      timeout: env.TIMEOUT,
      success(res) {
        const body = res.data || {}
        const requestId = res.header && (res.header['X-Request-Id'] || body.requestId)

        if (res.statusCode === 401) {
          if (!relogged) {
            // 静默重登后重放一次
            return relogin().then(t => {
              if (t) realRequest(opts, true, true).then(resolve, reject)
              else { require('./auth').handleUnauthorized(); reject(err('AUTH', 40101, '登录已失效', requestId)) }
            })
          }
          require('./auth').handleUnauthorized()
          return reject(err('AUTH', 40101, '登录已失效', requestId))
        }

        if (res.statusCode === 429) {
          const e = err('BIZ', 42901, '操作太频繁，请稍后再试', requestId)
          require('../utils/ui').toast(e.message)
          report.apiError(opts.url, e)
          return reject(e)
        }

        if (body.code === 0) {
          if (requestId) require('../store/store').ui.set({ lastRequestId: requestId })
          return resolve(body.data)
        }

        const e = err('BIZ', body.code, body.message, requestId)
        report.apiError(opts.url, e)
        reject(e)
      },
      fail(e) {
        const isGet = (opts.method || 'GET') === 'GET'
        if (!retried && isGet) {
          return setTimeout(() => realRequest(opts, true, relogged).then(resolve, reject), 800)
        }
        const ne = err('NETWORK', 50000, '网络开小差了', '')
        report.apiError(opts.url, ne)
        reject(ne)
      }
    })
  })
}

function request(opts) {
  const o = Object.assign({ method: 'GET', data: {}, auth: true }, opts || {})
  if (env.USE_MOCK) return mock.resolve(o.mockKey, o)

  const start = Date.now()
  return realRequest(o, false, false).then(
    r => { const c = Date.now() - start; if (c > 2000) log.warn('request', o.url, c + 'ms'); return r },
    e => { log.error('http', o.url, e.code, e.message); throw e }
  )
}

module.exports = { request, err, relogin }
```

### 4.3 关键约束

1. **重放只允许 1 次**（`relogged` 标志），避免死循环。
2. **并发去重**：`refreshing` Promise 共享，10 个并发 401 只重登 1 次。
3. **重放必须复用原 `opts`**，不得改写 body。
4. **不重试写操作**（POST/PUT/DELETE），避免重复发布/重复点赞。
5. `X-Request-Id` 由后端下发，前端只透传与展示，不生成。

## 5. 登录态与会话

```
                 ┌──────────────┐  token 存在且 A02 成功
   onLaunch ───► │  restoring   │ ─────────────────────► 已登录
                 └──────┬───────┘
                        │ 无 token / A02 失败
                        ▼
                 ┌──────────────┐  点登录/被拦截
                 │   未登录     │ ─────────────────────► G01
                 └──────┬───────┘                        │
                        │ A01 成功                        │ 取消
                        ▼                                ▼
                 ┌──────────────┐                     返回原页
                 │   已登录     │ ── 401 且重登失败 ──► 未登录
                 └──────────────┘
```

| 键 | 内容 | 有效期 |
|---|---|---|
| `jh_token` | JWT | 7 天（服务端 exp） |
| `jh_user` | 用户资料快照 | 跟随 token |
| `store.ui.pendingAction` | 登录后待执行动作 | 一次性 |

**规则**：登录成功后**先**刷新 `store.user`，**再**执行 `pendingAction`，**最后** `navigateBack`。

## 6. Store 规范

| store | 字段 | 谁写 | 谁读 |
|---|---|---|---|
| `user` | `token / isLogin / profile / completeness` | `services/auth` `store/user` | 全站 |
| `ui` | `tab / visited / pendingAction / loginJustNow / lastRequestId` | 页面 | 页面 |
| `feed` | `lists.{recommend,follow,latest}` | community-pane | community-pane、post 页（同步点赞态） |

**禁止**：把 20 条以上的列表放 store（内存与 diff 成本）；store 不持久化（除 token/user 经 `storage`）。

## 7. 组件规范

| 项 | 规定 |
|---|---|
| 命名 | 目录 kebab-case；组件名与目录同名；事件小驼峰 `bind:like`、`bind:tap` |
| 属性 | 复杂对象用单属性传入（`post="{{item}}"`），不要拆 10 个原子属性 |
| 事件 | 只抛语义事件（`like/collect/tap/author`），payload 最小化（`{id, liked}`） |
| 样式 | `options: { styleIsolation: 'isolated' }`；不使用 `page` 选择器；颜色用 CSS 变量 |
| 状态 | 组件内不请求、不读 store；需要数据由父级传 |
| 空值 | 所有图片走 `image-ph`；所有列表空走 `empty-state` |
| 复用 | 同一卡片在不同页面出现时，用 `mode` 属性区分，**不复制代码** |

## 8. 页面开发模板

```js
// packageXxx/pages/yyy/yyy.js
const mixin = require('../../../utils/page-mixin')
const api = require('../../../services/api')
const { ROUTES } = require('../../../config/constants')

Page(Object.assign({}, mixin.mixin, {
  data: Object.assign(mixin.baseData(), {
    loading: true,
    error: '',
    list: [],
    cursor: '',
    hasMore: true
  }),

  onLoad(options) {
    if (!options || !options.id) return this.failParam('缺少 id')
    this.id = options.id
    this.load(true)
  },

  onShow() { /* 需要刷新的页面在此做 60s 节流刷新 */ },

  async load(reset) {
    if (reset) this.setData({ loading: true, error: '' })
    try {
      const res = await api.getComments(this.id, reset ? '' : this.data.cursor, 20)
      this.setData({
        loading: false,
        list: reset ? res.list : this.data.list.concat(res.list),
        cursor: res.nextCursor || '',
        hasMore: !!res.hasMore
      })
    } catch (e) {
      this.setData({ loading: false, error: e.type === 'NETWORK' ? '网络开小差了' : e.message })
    }
  },

  onRetry() { this.load(true) }
}))
```

**规范**：`async/await` + try/catch；错误只落到 `data.error` 供 `empty-state` 使用；不在 catch 里裸 `console.log`（用 `utils/log`）。

## 10. 编码规范

| 项 | 规定 |
|---|---|
| 语法 | ES2018（`async/await`、模板串、可选链谨慎用）；不引入 Babel 插件 |

## 14. 常见坑（1.0 新增部分）

| # | 坑 | 处理 |
|---|---|---|
| 1 | 401 静默重登与 `handleUnauthorized` 互相触发 | 重登失败才跳登录，且 3 秒内只跳 1 次 |
| 2 | 重放时把已改的 body 再发一次导致重复发布 | 只重放 GET；写操作直接失败返回 |
| 3 | `wx.login` 频繁调用被限 | 重登加 1s 冷却，且只在 401 时触发 |
| 4 | 上传并发过高导致部分失败 | 并发 3，失败重试 1 次，已成功 URL 复用 |
| 5 | 头像上传后本地临时路径失效 | 上传成功立即用服务端 URL 覆盖 |
| 6 | `requestId` 未透传导致无法定位 | `X-Request-Id` 响应头优先，其次 `body.requestId` |
| 7 | Mock 与真实返回结构不一致 | Mock 必须按 `04` 的 DTO 字段生成，含 `requestId` 占位 |
| 8 | 生产包误带 mock 数据 | `env.prod` 分支内不 require mock；提测前 `check.js` 校验 |
